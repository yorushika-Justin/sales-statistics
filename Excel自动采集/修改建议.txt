全局进程退出钩子与孤儿进程治理针对 BUG-05，在主流程初始化完成浏览器启动后，注册全局系统退出事件，确保外部强行终止时底层浏览器能够得到彻底清理：JavaScriptfunction registerProcessGuards(browser) {
  const cleanUp = async () => {
    try {
      if (browser) {
        await browser.close();
        log('宿主机 Edge 浏览器进程已释放');
      }
    } catch (_) {}
    process.exit(1);
  };

  process.once('SIGINT', cleanUp);
  process.once('SIGTERM', cleanUp);
  process.once('SIGHUP', cleanUp);
}
